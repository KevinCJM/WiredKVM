cmake_minimum_required(VERSION 3.20)
project(WiredKVM_WindowsTray LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This tray prototype requires the Windows SDK and must be built on Windows.")
endif()
add_executable(WiredKVMTray WIN32 WiredKVMTray.cpp)
target_compile_features(WiredKVMTray PRIVATE cxx_std_17)
target_compile_definitions(WiredKVMTray PRIVATE UNICODE _UNICODE)
if(MSVC)
  target_compile_options(WiredKVMTray PRIVATE /W4 /utf-8)
endif()
target_link_libraries(WiredKVMTray PRIVATE user32 shell32)
